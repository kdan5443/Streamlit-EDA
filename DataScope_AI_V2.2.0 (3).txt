def detect_data_type(df: pd.DataFrame, time_col_name: str, entity_col_name: str):
    """Heuristically detects the data type: Time Series, Panel, or Cross-Sectional."""
    has_time_index = isinstance(df.index, (pd.DatetimeIndex, pd.PeriodIndex))

    if not has_time_index:
        return "Cross-Sectional", "No time index detected. Features requiring a time dimension will be unavailable."

    # Case 1: User specifies an entity column for Panel Data
    if entity_col_name:
        if entity_col_name not in df.columns:
            return "Time Series (Error)", f"Specified Entity ID column '{entity_col_name}' not found in data."
        # Check if the combination of the specified entity column and the time index is unique
        is_unique_panel = not df.reset_index().duplicated(subset=[df.index.name or 'index', entity_col_name]).any()
        if not is_unique_panel:
            return "Time Series (Panel Ambiguous)", f"The combination of time and '{entity_col_name}' is not unique. Treating as time series."
        return "Panel Data", f"Time index with specified entity ID '{entity_col_name}'."

    # Case 2: Time index has duplicates -> strongly suggests Panel Data is needed
    if not df.index.is_unique:
        return "Panel Data (Required)", "Time index has duplicates. Please specify an 'Entity ID column' to properly handle the data."

    # Case 3: Time index is unique -> likely a single Time Series
    return "Time Series", "Unique time index detected. Treating as a single time series."

@st.cache_data(show_spinner=False)
def _coerce_numeric(df: pd.DataFrame):
    df_num = pd.DataFrame(index=df.index)
    dropped = []
    for c in df.columns:
        if pd.api.types.is_numeric_dtype(df[c]):
            df_num[c] = df[c]
        else:
            x = pd.to_numeric(df[c], errors="coerce")
            if x.notna().sum() >= max(3, int(0.5 * len(x))):
                df_num[c] = x
            else:
                dropped.append(c)
    return df_num, dropped


def _validate_time_index(df: pd.DataFrame):
    msgs, ok = [], True
    if not isinstance(df.index, (pd.DatetimeIndex, pd.PeriodIndex)):
        msgs.append("Index is not datetime/period; Granger/Forecasting assume evenly-spaced rows.")
        ok = False
    else:
        if not df.index.is_monotonic_increasing:
            msgs.append("Datetime index not sorted; sorting by time.")
        if getattr(df.index, "inferred_freq", None) is None:
            msgs.append("Datetime frequency not inferred; ensure roughly even spacing for Granger/Forecasting.")
    return ok, msgs

@st.cache_data(show_spinner="Making series stationary…")
def compute_adf_stationary(df: pd.DataFrame, alpha: float):
    transformed, info = {}, {}
    for c in df.columns:
        s = df[c].dropna()
        if len(s) < 3:
            continue
        n_diffs = 0
        while True:
            try:
                p = adfuller(s)[1]
            except Exception:
                break
            if p <= alpha or n_diffs >= 2 or len(s) < 3:
                transformed[c], info[c] = s, n_diffs
                break
            s = s.diff().dropna(); n_diffs += 1
    if not transformed:
        return pd.DataFrame(index=df.index), {}
    return pd.concat(transformed, axis=1).dropna(), info


@st.cache_data(show_spinner=False)
def auto_select_lag(df: pd.DataFrame, max_lag_cap: int = 12, criterion: str = "aic"):
    try:
        model = VAR_MDL(df.dropna())
        res = model.select_order(maxlags=min(max_lag_cap, max(1, int(len(df) / 5))))
        k = getattr(res, criterion)
        return int(k) if k and not math.isnan(k) else 3
    except Exception:
        return 3


@st.cache_data(show_spinner="Running Granger causality tests…")
def granger_matrix(df: pd.DataFrame, max_lag: int, alpha: float, fdr: bool = True, search_best_lag: bool = False):
    cols, results = df.select_dtypes("number").columns.tolist(), []
    for cause in cols:
        for effect in cols:
            if cause == effect:
                continue
            data = df[[effect, cause]].dropna()
            if len(data) <= max_lag * 2: # ensure enough data
                continue
            try:
                lag_to_use = max_lag
                if search_best_lag:
                    best_p, best_l = 1.0, 1
                    for L in range(1, max_lag+1):
                        gc = grangercausalitytests(data, maxlag=L, verbose=False)
                        pL = gc[L][0]['ssr_ftest'][1]
                        if pL < best_p:
                            best_p, best_l = pL, L
                    lag_to_use = best_l
                gc = grangercausalitytests(data, maxlag=lag_to_use, verbose=False)
                f_stat, p = gc[lag_to_use][0]['ssr_ftest'][:2]
                results.append((cause, effect, lag_to_use, f_stat, p))
            except Exception:
                continue
    if not results:
        return pd.DataFrame(columns=["cause","effect","lag","F","p_raw","p_adj","significant"])
    pvals = [r[4] for r in results]
    rej, padj = (multipletests(pvals, alpha=alpha, method="fdr_bh")[0:2]) if fdr and len(pvals) > 1 else ([p < alpha for p in pvals], pvals)
    out = pd.DataFrame({
        "cause": [r[0] for r in results],
        "effect": [r[1] for r in results],
        "lag": [r[2] for r in results],
        "F": [r[3] for r in results],
        "p_raw": pvals,
        "p_adj": padj,
        "significant": rej,
    }).sort_values(["significant","p_adj","p_raw"], ascending=[False,True,True])
    return out


@st.cache_data(show_spinner="Inferring graph with PC…")
def pc_estimate(df: pd.DataFrame, alpha: float, ci_test: str = "pearsonr"):
    est = PC(data=df)
    model = est.estimate(ci_test=ci_test, significance_level=alpha)
    return nx.DiGraph(model.edges()), {'algo': 'PC (pgmpy)'}


@st.cache_data(show_spinner="Inferring PC equivalence class…")
def pc_estimate_equivalence_class(df: pd.DataFrame, alpha: float, ci_test: str = "pearsonr"):
    """
    Estimates the Markov Equivalence Class using the PC algorithm, 
    returning a graph with both directed and undirected edges.
    """
    est = PC(data=df)
    # Estimate the CPDAG (Pattern) which represents the equivalence class
    model_cpdag = est.estimate(ci_test=ci_test, significance_level=alpha, return_type="cpdag")
    
    # Create a graph to hold the results
    G = nx.DiGraph()
    G.add_nodes_from(model_cpdag.nodes())
    
    # Add directed edges (certain direction)
    for u, v in model_cpdag.directed_edges:
        G.add_edge(u, v, type='directed')
        
    # Add undirected edges (ambiguous direction)
    for u, v in model_cpdag.undirected_edges:
        G.add_edge(u, v, type='undirected')
        G.add_edge(v, u, type='undirected') # Add reverse for plotting symmetry

    return G, {'algo': 'PC Equivalence Class (pgmpy)'}


@st.cache_data(show_spinner="Inferring graph with GES…")
def ges_estimate(df: pd.DataFrame):
    data, var_names = df.values, list(df.columns)
    res = cl_ges.ges(data, scoring_func='local_score_BIC')
    dag = nx.DiGraph(); dag.add_nodes_from(var_names)
    try:
        mat = res['G'].graph
        for i in range(mat.shape[0]):
            for j in range(mat.shape[1]):
                if mat[i, j] == 2:
                    dag.add_edge(var_names[i], var_names[j])
    except Exception:
        pass
    return dag, {'algo': 'GES (causal-learn)'}

def notears_estimate(df: pd.DataFrame):

@st.cache_data(show_spinner="Inferring graph with LiNGAM…")
def lingam_estimate(df: pd.DataFrame):
    mdl = lingam.DirectLiNGAM(random_state=GLOBAL_SEED); mdl.fit(df.values)
    B, cols = mdl.adjacency_matrix_, list(df.columns)
    dag = nx.DiGraph(); dag.add_nodes_from(cols)
    for i in range(B.shape[0]):
        for j in range(B.shape[1]):
            if abs(B[i, j]) > 1e-8:
                dag.add_edge(cols[j], cols[i])
    return dag, {'algo': 'DirectLiNGAM'}


def _pick_autodag(df, backend: str = "auto", alpha: float = 0.05):
    dag, meta, used = nx.DiGraph(), {}, None
    try:
        if backend == "auto":
            if notears_from_pandas:
                backend = "notears"
            else:
                backend = "pc"
        if backend == "notears":
            dag, meta = notears_estimate(df); used = "NOTEARS"
        elif backend == "pc":
            dag, meta = pc_estimate(df, alpha=alpha); used = "PC"
        elif backend == "ges":
            if cl_ges is None:
                raise RuntimeError("causal-learn not installed")
            dag, meta = ges_estimate(df); used = "GES"
        elif backend == "lingam":
            if lingam is None:
                raise RuntimeError("lingam not installed")
            dag, meta = lingam_estimate(df); used = "LiNGAM"
    except Exception as e:
        meta = {"error": str(e)}; used = used or backend
    return dag, meta, used


@st.cache_data
def detect_seasonality(series: pd.Series):
    """Detects seasonality period using ACF. Handles non-datetime indexes."""
    series = series.dropna()
    min_len = 20

    if not isinstance(series.index, (pd.DatetimeIndex, pd.PeriodIndex)):
        return 0, "Numeric Index"

    if len(series) < min_len:
        return 0, None

    freq = pd.infer_freq(series.index)
    plausible_m = []
    if freq:
        if freq.startswith('D'): plausible_m.append(7)
        elif freq.startswith('W'): plausible_m.append(52)
        elif freq.startswith('M'): plausible_m.append(12)
        elif freq.startswith('Q'): plausible_m.append(4)

    plausible_m.extend([m for m in [4, 7, 12, 52] if m not in plausible_m])

    max_lags = min(len(series) // 2, 100)
    try:
        acf_vals = acf(series, nlags=max_lags, fft=True)
    except Exception:
        return 0, freq

    best_m, best_corr = 0, 0.0
    for m in plausible_m:
        if m < len(acf_vals) and m > 1:
            current_corr = acf_vals[m]
            if current_corr > best_corr and current_corr > 0.3:
                is_peak = True
                if m > 1 and m < len(acf_vals) - 1:
                    if not (current_corr > acf_vals[m-1] and current_corr > acf_vals[m+1]):
                        is_peak = False
                if is_peak:
                    best_corr = current_corr
                    best_m = m
    return best_m, freq

# --- Helper functions to classify node roles in the graph ---
def classify_nodes(dag: nx.DiGraph):
    """Classifies nodes for AutoDAG graphs (handles CPDAGs)."""
    node_types = {}
    directed_edges = [(u, v) for u, v, data in dag.edges(data=True) if data.get('type') == 'directed']
    
    temp_dag = nx.DiGraph(directed_edges)
    temp_dag.add_nodes_from(dag.nodes()) 

    for node in dag.nodes():
        in_degree = temp_dag.in_degree(node)
        out_degree = temp_dag.out_degree(node)
        is_in_undirected = any(data.get('type') == 'undirected' for _, _, data in dag.edges(node, data=True))

        if in_degree == 0 and out_degree > 0 and not is_in_undirected:
            node_types[node] = 'root'
        elif out_degree == 0 and in_degree > 0 and not is_in_undirected:
            node_types[node] = 'effect'
        else:
            node_types[node] = 'mediator'
    return node_types

def classify_granger_nodes(G: nx.DiGraph):
    """Classifies nodes for the purely directed Granger graph."""
    node_types = {}
    for node in G.nodes():
        in_degree = G.in_degree(node)
        out_degree = G.out_degree(node)
        if in_degree == 0 and out_degree > 0:
            node_types[node] = 'root' # Granger Driver
        elif out_degree == 0 and in_degree > 0:
            node_types[node] = 'effect' # Granger Outcome
        else:
            node_types[node] = 'mediator'
    return node_types

# NEW HELPER
def classify_summary_nodes(G: nx.DiGraph):
    """Classifies nodes for the combined summary graph based on directed edges."""
    node_types = {}
    # Create a subgraph with only the directed edges to calculate degrees accurately
    directed_G = nx.DiGraph([(u, v) for u, v, data in G.edges(data=True) if data.get('type') == 'directed'])
    directed_G.add_nodes_from(G.nodes()) # Ensure all nodes are included

    for node in G.nodes():
        in_degree = directed_G.in_degree(node)
        out_degree = directed_G.out_degree(node)
        
        if in_degree == 0 and out_degree > 0:
            node_types[node] = 'Driver'
        elif out_degree == 0 and in_degree > 0:
            node_types[node] = 'Outcome'
        elif in_degree > 0 and out_degree > 0:
            node_types[node] = 'Mediator'
        else: # in_degree == 0 and out_degree == 0 but may have undirected edges
            node_types[node] = 'Isolated'
    return node_types

def fig_cpdag_plotly(dag: nx.DiGraph, node_types: dict, title="Causal Graph"):
    """
    Visualizes a causal graph, coloring nodes by role and styling edges by type.
    Handles both regular DAGs and CPDAGs (from PC Equivalence Class).
    """
    if dag is None or dag.number_of_nodes() == 0: return go.Figure()
    try:
        pos = nx.drawing.nx_pydot.graphviz_layout(dag, prog="dot")
    except Exception:
        pos = nx.spring_layout(dag, k=0.9, iterations=50, seed=GLOBAL_SEED)

    # Edge traces
    edge_x_dir, edge_y_dir, edge_x_undir, edge_y_undir = [], [], [], []
    for u, v, data in dag.edges(data=True):
        # All edges from non-PC methods are treated as 'directed'
        edge_type = data.get('type', 'directed')

        if edge_type == 'directed':
            edge_x_dir.extend([pos[u][0], pos[v][0], None])
            edge_y_dir.extend([pos[u][1], pos[v][1], None])
        # To avoid drawing undirected edges twice, we only draw from the 'smaller' node name
        elif edge_type == 'undirected' and u < v:
            edge_x_undir.extend([pos[u][0], pos[v][0], None])

def _infer_step(idx):
    if isinstance(idx, pd.PeriodIndex): return idx.freq
    if isinstance(idx, pd.DatetimeIndex):
        if idx.freq: return idx.freq
        f = pd.infer_freq(idx)
        if f: return f
        if len(idx) >= 2: return idx[-1] - idx[-2]
        return pd.Timedelta(days=1)
    return 1


def _future_index(idx, steps):
    if isinstance(idx, (pd.DatetimeIndex, pd.PeriodIndex)):
        try:
            step = _infer_step(idx)
            start = idx[-1] + step
            return pd.date_range(start=start, periods=steps, freq=step)
        except Exception:
            last_date = idx[-1]
            time_deltas = np.diff(idx.to_pydatetime())
            avg_delta = np.mean(time_deltas)
            return pd.to_datetime([last_date + (i+1)*avg_delta for i in range(steps)])

    last = idx[-1] if len(idx) > 0 else -1
    return np.arange(last + 1, last + 1 + steps)

def create_lagged_features(data, n_lags=5):
    df = pd.DataFrame(data)
    cols = []
    for i in range(1, n_lags + 1):
        cols.append(df.shift(i))
    agg = pd.concat(cols, axis=1).dropna()
    agg.columns = [f'lag_{i}' for i in range(1, n_lags + 1)]
    return agg.values, df.iloc[n_lags:].values.ravel()

@st.cache_data
def geometric_adstock(series, theta):
    """Applies geometric adstock transformation."""
    if theta == 0:
        return series
    adstocked_series = np.zeros_like(series, dtype=float)
    adstocked_series[0] = series[0]
    for i in range(1, len(series)):
        adstocked_series[i] = series[i] + theta * adstocked_series[i-1]
    return adstocked_series

@st.cache_data
def hill_saturation(series, alpha, gamma):
    """Applies Hill saturation transformation."""
    return series**alpha / (series**alpha + gamma**alpha)

class TCN(nn.Module):
    def __init__(self, input_size, output_size, num_channels, kernel_size, dropout):
        super(TCN, self).__init__(); self.conv1 = nn.Conv1d(input_size, num_channels, kernel_size, padding=(kernel_size - 1)); self.relu1 = nn.ReLU(); self.dp1 = nn.Dropout(dropout); self.conv2 = nn.Conv1d(num_channels, num_channels, kernel_size, padding=(kernel_size - 1)); self.relu2 = nn.ReLU(); self.dp2 = nn.Dropout(dropout); self.linear = nn.Linear(num_channels, output_size)
    def forward(self, x):
        y = self.dp1(self.relu1(self.conv1(x)))[:, :, :-self.conv1.padding[0]]; y = self.dp2(self.relu2(self.conv2(y)))[:, :, :-self.conv2.padding[0]]; return self.linear(y[:,:,-1])

def create_sequences(data, seq_length):
    xs, ys = [], [];
    for i in range(len(data)-seq_length): xs.append(data.iloc[i:(i+seq_length)]); ys.append(data.iloc[i+seq_length])
    return np.array(xs), np.array(ys)

@st.cache_data(show_spinner="Running Granger causality for summary...")
def run_granger_causality_summary(df: pd.DataFrame, max_lag: int, alpha: float = 0.05):
    """Computes Granger causality, finding the best lag for each pair."""
    cols = df.columns
    results = []
    for cause in cols:
        for effect in cols:
            if cause == effect: continue
            data = df[[effect, cause]].dropna()
            if len(data) < max_lag * 3: continue
            try:
                best_p, best_lag = 1.0, 1
                for lag in range(1, max_lag + 1):
                    gc_res = grangercausalitytests(data, [lag], verbose=False)
                    p_val = gc_res[lag][0]['ssr_ftest'][1]
                    if p_val < best_p:
                        best_p = p_val
                        best_lag = lag
                
                final_gc = grangercausalitytests(data, [best_lag], verbose=False)
                _, p_val = final_gc[best_lag][0]['ssr_ftest'][:2]
                results.append((cause, effect, best_lag, p_val))
            except Exception:
                continue
    
    if not results: return pd.DataFrame()
    pvals = [r[3] for r in results]
    rejected, p_adj, _, _ = multipletests(pvals, alpha=alpha, method='fdr_bh')
    out = pd.DataFrame({
        "Cause": [r[0] for r in results], "Effect": [r[1] for r in results],
        "Lag": [r[2] for r in results], "P-Value (adj)": p_adj, "Significant": rejected
    }).sort_values("P-Value (adj)")
    return out

@st.cache_data(show_spinner="Inferring causal skeleton with PC algorithm...")
def run_pc_equivalence_for_summary(df: pd.DataFrame, alpha: float = 0.05):
    """Estimates the Markov Equivalence Class using the PC algorithm."""
    est = PC(data=df.dropna())
    model = est.estimate(ci_test="pearsonr", significance_level=alpha, return_type="cpdag")
    G = nx.DiGraph()
    G.add_nodes_from(model.nodes())
    for u, v in model.directed_edges: G.add_edge(u, v, type='directed', lag=0)
    for u, v in model.undirected_edges: G.add_edge(u, v, type='undirected', lag=0)
    return G
